cmake_minimum_required(VERSION 3.16)
project(mnn_digits CXX)
set(CMAKE_CXX_STANDARD 17)
set(MNN_INCLUDE "" CACHE PATH "MNN include directory")
set(MNN_LIB "" CACHE FILEPATH "libMNN.so")
add_executable(mnn_digits main.cpp)
target_include_directories(mnn_digits PRIVATE ${MNN_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR}/../common/cpp)
if(MNN_LIB MATCHES "\\.a$")
  # Static MNN: backends register themselves in static initializers, so the archive must be linked whole.
  target_link_libraries(mnn_digits PRIVATE -Wl,--whole-archive ${MNN_LIB} -Wl,--no-whole-archive)
else()
  target_link_libraries(mnn_digits PRIVATE ${MNN_LIB})
endif()
if(ANDROID)
  target_link_libraries(mnn_digits PRIVATE log)
endif()
get_filename_component(MNN_LIB_DIR ${MNN_LIB} DIRECTORY)
set_target_properties(mnn_digits PROPERTIES BUILD_RPATH "\$ORIGIN;${MNN_LIB_DIR}")
