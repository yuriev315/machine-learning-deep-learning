cmake_minimum_required(VERSION 3.16)
project(ort_digits CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORT_ROOT "" CACHE PATH "Extracted onnxruntime-linux-<arch>-1.22.0 directory")
if(NOT ORT_ROOT)
  message(FATAL_ERROR "Pass -DORT_ROOT=<path to onnxruntime SDK>")
endif()
add_executable(ort_digits main.cpp)
target_include_directories(ort_digits PRIVATE ${ORT_ROOT}/include ${CMAKE_CURRENT_SOURCE_DIR}/../common/cpp)
target_link_libraries(ort_digits PRIVATE ${ORT_ROOT}/lib/libonnxruntime.so)
# Find libonnxruntime next to the binary or in ../lib, so the folder can be copied to a device as is.
set_target_properties(ort_digits PROPERTIES BUILD_RPATH "\$ORIGIN;\$ORIGIN/../lib;${ORT_ROOT}/lib" INSTALL_RPATH "\$ORIGIN/../lib")
install(TARGETS ort_digits RUNTIME DESTINATION bin)
