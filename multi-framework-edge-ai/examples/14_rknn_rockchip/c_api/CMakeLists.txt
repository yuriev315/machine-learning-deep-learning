cmake_minimum_required(VERSION 3.16)
project(rknn_digits CXX)
set(CMAKE_CXX_STANDARD 17)
set(RKNN_API_INCLUDE "" CACHE PATH "Directory with rknn_api.h")
set(RKNN_RT_LIB "" CACHE FILEPATH "aarch64 librknnrt.so")
add_executable(rknn_digits main.cpp)
target_include_directories(rknn_digits PRIVATE ${RKNN_API_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR}/../../common/cpp)
target_link_libraries(rknn_digits PRIVATE ${RKNN_RT_LIB})
set_target_properties(rknn_digits PROPERTIES BUILD_RPATH "\$ORIGIN")
