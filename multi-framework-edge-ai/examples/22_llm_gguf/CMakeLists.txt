cmake_minimum_required(VERSION 3.16)
project(llm_gguf CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_SRC "" CACHE PATH "llama.cpp source")
set(LLAMA_BUILD "" CACHE PATH "llama.cpp build dir")
add_executable(llm_gguf main.cpp)
target_include_directories(llm_gguf PRIVATE ${LLAMA_SRC}/include ${LLAMA_SRC}/ggml/include)
target_link_libraries(llm_gguf PRIVATE ${LLAMA_BUILD}/bin/libllama.so ${LLAMA_BUILD}/bin/libggml.so ${LLAMA_BUILD}/bin/libggml-base.so ${LLAMA_BUILD}/bin/libggml-cpu.so)
set_target_properties(llm_gguf PROPERTIES BUILD_RPATH "${LLAMA_BUILD}/bin")
