cmake_minimum_required(VERSION 3.16)
project(litert_digits C CXX)
set(CMAKE_CXX_STANDARD 17)
set(TFLITE_SRC "" CACHE PATH "TensorFlow source root (headers)")
set(TFLITE_LIB "" CACHE FILEPATH "Path to libtensorflowlite_c.so")
if(NOT TFLITE_SRC OR NOT TFLITE_LIB)
  message(FATAL_ERROR "Pass -DTFLITE_SRC and -DTFLITE_LIB")
endif()
add_executable(litert_digits main.cpp)
target_include_directories(litert_digits PRIVATE ${TFLITE_SRC} ${CMAKE_CURRENT_SOURCE_DIR}/../common/cpp)
target_link_libraries(litert_digits PRIVATE ${TFLITE_LIB})
get_filename_component(TFLITE_LIB_DIR ${TFLITE_LIB} DIRECTORY)
set_target_properties(litert_digits PROPERTIES BUILD_RPATH "\$ORIGIN;${TFLITE_LIB_DIR}")
